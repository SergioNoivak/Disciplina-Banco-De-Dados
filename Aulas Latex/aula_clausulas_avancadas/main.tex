\section{Integridade Referencial}

\begin{frame}{O que é Integridade Referencial? (Teoria)}
  \begin{block}{Definição}
    É um conjunto de regras que garante que a relação entre registros de tabelas diferentes permaneça válida e consistente. No modelo relacional, isso significa que uma referência de uma tabela para outra deve sempre apontar para um dado existente.
  \end{block}
  \vspace{0.4cm}
  \begin{itemize}
    \item \textbf{Consistência}: Garante que não existam referências a dados inexistentes.
    \item \textbf{Confiabilidade}: O banco de dados impede ações que quebrem esses vínculos.
    \item \textbf{Automação}: O SGDB cuida da fiscalização, poupando o trabalho da aplicação.
  \end{itemize} 
\end{frame}

\begin{frame}{O Papel da Chave Estrangeira (Teoria)}
  \begin{itemize}
    \item A \textbf{Chave Estrangeira (Foreign Key)} é o instrumento técnico da integridade referencial.
    \item Ela é uma coluna (ou conjunto de colunas) em uma tabela que armazena a \textbf{Chave Primária} de outra tabela.
  \end{itemize}
  \vspace{0.3cm}
  \centering
  \begin{tikzpicture}[node distance=1cm, font=\small\sffamily]
    \node (parent) [draw, rectangle, fill=blue!10, text width=3cm, align=center, thick] {
      \textbf{PAI} \\ \rule{2.8cm}{0.4pt} \\
      \underline{PK (ID)}
    };
    \node (child) [draw, rectangle, fill=green!10, text width=3cm, align=center, right=2cm of parent, thick] {
      \textbf{FILHO} \\ \rule{2.8cm}{0.4pt} \\
      FK (ID\_PAI)
    };
    \draw[->, >=stealth, ultra thick, structure] (child.west) -- (parent.east) node[midway, above] {Aponte para};
  \end{tikzpicture}
  \vspace{0.3cm}
  \begin{alertblock}{Regra de Ouro}
    O valor na FK deve existir na PK da tabela referenciada ou ser nulo (se a regra do negócio permitir).
  \end{alertblock}
\end{frame}

\begin{frame}{Restrições no Modelo Relacional (Teoria)}
  \begin{block}{O Papel das Constraints}
    As restrições de integridade referencial funcionam como "guardiões" do banco de dados, impedindo que qualquer operação (Manual ou via Aplicação) deixe os dados em um estado inconsistente.
  \end{block}
  \vspace{0.4cm}
  \begin{itemize}
    \item Elas são verificadas \textbf{instantaneamente} pelo SGDB no momento da execução do comando.
    \item Se a regra for violada, a transação é abortada e o erro é retornado.
    \item Garante que a "verdade" dos dados seja mantida sem depender da lógica do programador.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Restrição de Inserção (Prática)}
  \small
  Não é permitido criar um registro "filho" que aponte para um "pai" inexistente.
  \vspace{0.3cm}
  \centering
  \begin{tikzpicture}[font=\scriptsize\sffamily]
    \node (p) [draw, fill=blue!10, text width=2cm, align=center] {\textbf{Clientes}\\ID: 1\\ID: 2};
    \node (f) [draw, fill=green!10, text width=3cm, align=center, right=2cm of p] {\textbf{Novo Pedido}\\ID\_Cliente: \color{red}999};
    
    \draw[->, ultra thick, red] (f.west) -- (p.east) node[midway, above] {ID não existe!};
    \node at ($(p.east)!0.5!(f.west)$) [draw, circle, red, ultra thick, cross out, minimum size=0.5cm] {};
  \end{tikzpicture}\\
  \centering\scriptsize
  \textbf{Clientes}(\underline{id}, nome) \quad \textbf{Pedidos}(\underline{id}, \color{red}id\_cliente\color{black}, valor) \\
  \textit{id\_cliente referencia Clientes}
  \vspace{0.2cm}
  \begin{exampleblock}{Tentativa de SQL}
\begin{lstlisting}
-- O Cliente 999 não existe na tabela Clientes

VALUES (101, 250.00, 999);
\end{lstlisting}
    \vspace{0.1cm}
    \color{red}\textit{Erro: insert or update on table "pedidos" violates foreign key constraint...}
  \end{exampleblock}
\end{frame}

  \end{tikzpicture}
  \centering\scriptsize
  \textbf{Clientes}(\underline{id}, nome) \quad \textbf{Pedidos}(\underline{id}, \color{red}id\_cliente\color{black}, valor) \\
  \textit{id\_cliente referencia Clientes}
  \vspace{0.2cm}
  \begin{exampleblock}{Tentativa de SQL}
\begin{lstlisting}
-- Tentar deletar um cliente que já tem compras
DELETE FROM Clientes WHERE id = 1;
\end{lstlisting}
    \vspace{0.1cm}
    \color{red}\textit{Erro: update or delete on table "clientes" violates foreign key constraint...}
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Restrição de Alteração (Prática)}
  \small
  Alterar a Chave Primária do pai "quebraria" o vínculo com os filhos já existentes.
  \vspace{0.3cm}
  \centering
  \begin{tikzpicture}[font=\scriptsize\sffamily]

    \node (f) [draw, fill=green!10, text width=2.5cm, align=center, right=2cm of p] {\textbf{Item\_Venda}\\FK\_Prod: 'T100'};

    \node (upd) [above=0.3cm of p, structure, font=\bfseries] {UPDATE PK para 'T200'};
    \draw[->, bend right, red, ultra thick] (upd) -- (p);

  \end{tikzpicture}
  \centering\scriptsize
  \textbf{Produtos}(\underline{id}, nome) \quad \textbf{Item\_Venda}(\underline{id}, \color{red}id\_prod\color{black}) \\
  \textit{id\_prod referencia Produtos}
  \vspace{0.2cm}
  \begin{exampleblock}{Tentativa de SQL}
\begin{lstlisting}
-- Tentar mudar o código do produto que já foi vendido
UPDATE Produtos SET id = 'T200' WHERE id = 'T100';
\end{lstlisting}
    \vspace{0.1cm}
    \color{red}\textit{Erro: update or delete on table "produtos" violates foreign key constraint...}

\end{frame}

\begin{frame}{Consistência de Domínios e Tipos (Teoria)}
  \begin{itemize}
    \item Para que a integridade funcione, o \textbf{Domínio} (tipo de dado) da FK deve ser idêntico ao da PK.
    \item Se a PK é \texttt{INT}, a FK deve ser \texttt{INT}.
    \item Se a PK é \texttt{UUID}, a FK deve ser \texttt{UUID}.
  \end{itemize}
  \vspace{0.4cm}
  \begin{block}{Por que isso importa?}
    Tipos incompatíveis impedem que o SGDB realize as verificações de igualdade necessárias para validar o vínculo, além de causarem erros graves de performance em \textit{JOINs}.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Problema 1: Registros Órfãos (Integridade)}
  \small
  Um registro órfão ocorre quando um "filho" referencia um "pai" que não existe mais.
  \vspace{0.2cm}
  \centering
  \begin{tikzpicture}[font=\tiny\sffamily, node distance=1.5cm]
    \node (p) [draw, fill=blue!10, text width=2cm, align=center] {\textbf{Clientes}\\ID: 1\\ID: 2\\ \color{gray}...};
    \node (f) [draw, fill=green!10, text width=2cm, align=center, right=2.5cm of p] {\textbf{Pedidos}\\ID: 101, FK: 1\\ID: 102, FK: \textbf{\color{red}500}};
    
    \draw[->, thick] (f.west |- 0,-0.1) -- (p.east |- 0,-0.1);
    \draw[->, thick, red, dashed] (f.west |- 0,-0.4) -- +( -1.5,0) node[above] {Cadê o 500?};
    \node at (p.east) [xshift=1.2cm, yshift=-0.4cm, red, font=\huge] {?};
  \end{tikzpicture}
  \centering\scriptsize
  \textbf{Clientes}(\underline{id}, nome) \quad \textbf{Pedidos}(\underline{id}, \color{red}id\_cliente\color{black}, valor) \\
  \textit{id\_cliente referencia Clientes}
  \vspace{0.1cm}
  \begin{exampleblock}{Como detectar órfãos (SQL)}
\begin{lstlisting}
SELECT p.id, p.id_cliente FROM Pedidos p
LEFT JOIN Clientes c ON p.id_cliente = c.id
WHERE c.id IS NULL;
\end{lstlisting}
  \end{exampleblock}
  \begin{alertblock}{Consequências}
    Relatórios de vendas inconsistentes, erros de \textit{NullPointer} na aplicação e perda da integridade lógica do negócio.
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Problema 2: Inconsistência em Exclusões (Integridade)}
  \small
  Ao excluir um registro pai sem proteção, os filhos perdem a referência e tornam-se inválidos.
  \vspace{0.2cm}
  \centering
  \begin{tikzpicture}[font=\tiny\sffamily]
    \node (p) [draw, fill=red!20, text width=2cm, align=center] {\textbf{Clientes}\\ \sout{ID: 1}\\ID: 2};
    \node (f) [draw, fill=green!10, text width=2.5cm, align=center, right=2cm of p] {\textbf{Pedidos}\\ID: 101, FK: \textbf{\color{red}1} $\leftarrow$ ?\\ID: 102, FK: 2};
    
    \draw[->, red, thick, dashed] (f.west |- 0,0.1) -- (p.east |- 0,0.1);
    \node at (p) [draw, circle, red, ultra thick, cross out, minimum size=0.8cm] {};
  \end{tikzpicture}
  \centering\scriptsize
  \textbf{Clientes}(\underline{id}, nome) \quad \textbf{Pedidos}(\underline{id}, \color{red}id\_cliente\color{black}, valor) \\
  \textit{id\_cliente referencia Clientes}
  \vspace{0.1cm}
  \begin{exampleblock}{Exemplo SQL (Sem Constraints)}
\begin{lstlisting}
DELETE FROM Clientes WHERE id = 1;
-- Agora a consulta abaixo retorna dados sem sentido:
SELECT * FROM Pedidos WHERE id_cliente = 1;
\end{lstlisting}
  \end{exampleblock}
\end{frame}

  \small
  Alterar a Chave Primária de um pai invalida instantaneamente todos os filhos.
  \vspace{0.2cm}
  \centering
  \begin{tikzpicture}[font=\tiny\sffamily]
    \node (p) [draw, fill=blue!10, text width=2.5cm, align=center] {\textbf{Produtos}\\ID: 'P1' $\rightarrow$ \textbf{'P99'}};
    \node (f) [draw, fill=green!10, text width=2.5cm, align=center, right=2cm of p] {\textbf{Estoque}\\Prod\_ID: 'P1' \textbf{\color{red}(Quebrado)}};
    
    \draw[->, red, thick, dashed] (f.west) -- (p.east);
    \node at ($(p.east)!0.5!(f.west)$) [draw, circle, red, ultra thick, cross out, minimum size=0.5cm] {};
  \end{tikzpicture}
  \centering\scriptsize
  \textbf{Produtos}(\underline{id}, nome) \quad \textbf{Estoque}(\underline{id\_prod}, qtd) \\
  \textit{id\_prod referencia Produtos}
  \vspace{0.1cm}
  \begin{exampleblock}{Exemplo SQL (Sem ON UPDATE CASCADE)}
\begin{lstlisting}
UPDATE Produtos SET id = 'P99' WHERE id = 'P1';
-- Os registros de estoque continuam apontando para 'P1',
-- que não existe mais na tabela de produtos.
\end{lstlisting}
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Problema 4: Dependências Circulares (Integridade)}
  \small
  Ocorre quando duas tabelas dependem uma da outra para existir.
  \vspace{0.4cm}
  \centering
  \begin{tikzpicture}[node distance=2cm, font=\tiny\sffamily]
    \node (a) [draw, rectangle, fill=blue!10, text width=2cm, align=center] {\textbf{Tabela A}\\FK\_B (NOT NULL)};
    \node (b) [draw, rectangle, fill=green!10, right=of a, text width=2cm, align=center] {\textbf{Tabela B}\\FK\_A (NOT NULL)};
    
    \draw[->, bend left, thick, red] (a) edge node[above] {Precisa de B} (b);
    \draw[->, bend left, thick, red] (b) edge node[below] {Precisa de A} (a);
  \end{tikzpicture}
  \centering\scriptsize
  \textbf{Tabela\_A}(\underline{id}, \color{red}fk\_b\color{black}) \quad \textbf{Tabela\_B}(\underline{id}, \color{red}fk\_a\color{black}) \\
  \textit{fk\_b referencia B} \quad \textit{fk\_a referencia A}
  \vspace{0.2cm}
  \begin{exampleblock}{O Impasse do SQL}
\begin{lstlisting}
INSERT INTO A (id, fk_b) VALUES (1, 10);
-- Erro: B 10 não existe!
INSERT INTO B (id, fk_a) VALUES (10, 1);
-- Erro: A 1 não existe!
\end{lstlisting}
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Solução 1: Permitindo Nulos e Atualização Tardia}
  \small
  A maneira mais simples de resolver o impasse é permitir que uma das chaves estrangeiras aceite valores \textbf{NULL} temporariamente, quebrando a obrigatoriedade inicial.
  \vspace{0.2cm}
  \begin{exampleblock}{Passo a Passo da Inserção}
\begin{lstlisting}
-- 1. Inserir na Tabela A deixando a FK nula (fk_b deve ser NULL)
INSERT INTO A (id, fk_b) VALUES (1, NULL);

-- 2. Inserir na Tabela B referenciando o ID de A
INSERT INTO B (id, fk_a) VALUES (10, 1);

-- 3. Atualizar a Tabela A com o ID de B recém-criado
UPDATE A SET fk_b = 10 WHERE id = 1;
\end{lstlisting}
  \end{exampleblock}
  \begin{alertblock}{Vantagem}
    É a solução mais compatível entre diferentes SGDBs e fácil de entender logicamente.
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Solução 2: Restrições Postergáveis (DEFERRABLE)}
  \small
  Alguns bancos (PostgreSQL, Oracle, SQLite) permitem configurar a restrição para ser verificada apenas no \textbf{COMMIT} da transação.
  \vspace{0.2cm}
  \begin{exampleblock}{Configuração e Uso (Exemplo Conceitual)}
\begin{lstlisting}
-- Definir a FK como postergável na criação da tabela
CREATE TABLE A (
  id INT PRIMARY KEY,
  fk_b INT REFERENCES B(id) DEFERRABLE INITIALLY DEFERRED
);

-- Realizar as inserções dentro de uma transação
BEGIN TRANSACTION;
  INSERT INTO A (id, fk_b) VALUES (1, 10);
  INSERT INTO B (id, fk_a) VALUES (10, 1);
COMMIT; -- O banco valida tudo somente neste momento!
\end{lstlisting}
  \end{exampleblock}
  \textit{*Nota: No SQLite, use \texttt{PRAGMA defer\_foreign\_keys = ON;} dentro da transação.}
\end{frame}

\begin{frame}[fragile]{Solução 2: Exemplo Prático no SQLite}
  \small
  No SQLite, podemos adiar a verificação mesmo que a tabela não tenha sido criada com a cláusula \texttt{DEFERRABLE}, usando um \textbf{PRAGMA} especial.
  \vspace{0.2cm}
  \begin{exampleblock}{Uso do PRAGMA defer\_foreign\_keys}
\begin{lstlisting}
-- 1. Garante que as chaves estrangeiras estao ativas
PRAGMA foreign_keys = ON;

BEGIN TRANSACTION;
  -- 2. Forca o adiamento das verificacoes nesta transacao
  PRAGMA defer_foreign_keys = ON; 

  INSERT INTO Tabela_A (id, fk_b) VALUES (1, 10); -- Sem erro!
  INSERT INTO Tabela_B (id, fk_a) VALUES (10, 1);
COMMIT; -- A integridade e validada aqui.
\end{lstlisting}
  \end{exampleblock}
  \begin{alertblock}{Dica}
    O \texttt{PRAGMA defer\_foreign\_keys} é redefinido automaticamente para \texttt{OFF} assim que a transação termina (\texttt{COMMIT} ou \texttt{ROLLBACK}).
  \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Problema 5: Performance e Índices (Integridade)}
  \small
  Verificar a integridade em tabelas gigantescas sem índices causa lentidão extrema.
  \vspace{0.2cm}
  \centering
  \begin{tikzpicture}[font=\tiny\sffamily]
    \node (table) [draw, fill=gray!10, text width=4cm, align=center] {Tabela Pedidos\\(10 Milhões de Linhas)};
    \node (scan) [below=0.5cm of table, red] {Busca Sequencial Lenta...};
    \foreach \x in {0.1,0.3,...,3.9} \draw[red] (\x-2,-0.1) -- (\x-2,-0.4);
    
    \node (sol) [right=1.5cm of table, draw, fill=green!20, text width=3cm, align=center] {Busca via Índice\\(Instantânea)};
    \draw[->, structure, ultra thick] (table) -- (sol);
  \end{tikzpicture}
  \vspace{0.4cm}
  \begin{exampleblock}{Solução SQL}
\begin{lstlisting}
-- Sempre indexar as chaves estrangeiras:
CREATE INDEX idx_pedidos_cliente ON Pedidos(id_cliente);
\end{lstlisting}
  \end{exampleblock}
  \centering\scriptsize
  \textbf{Pedidos}(\underline{id}, \color{red}id\_cliente\color{black}, valor) \quad \textit{id\_cliente referencia Clientes}
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Dica}: A maioria dos bancos não cria índices automáticos para FKs!
  \end{itemize}
\end{frame}

\begin{frame}{Ações Referenciais: ON DELETE e ON UPDATE (Teoria)}
  \begin{block}{O que são?}
    São gatilhos automáticos definidos na criação da Tabela que determinam como o banco de dados deve reagir quando um registro referenciado (Pai) sofre uma alteração ou exclusão.
  \end{block}
  \vspace{0.3cm}
  \begin{itemize}
    \item \textbf{ON DELETE}: Define o comportamento na exclusão do pai.
    \item \textbf{ON UPDATE}: Define o comportamento na alteração da PK do pai.
  \end{itemize}
  \vspace{0.3cm}
  \begin{exampleblock}{Ações Possíveis}
    \lstinline|CASCADE, SET NULL, SET DEFAULT, RESTRICT, NO ACTION|
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{Ações Referenciais: CASCADE (Prática)}
  \small
  A ação \textbf{CASCADE} propaga a alteração do pai para todos os filhos vinculados.
  \vspace{0.3cm}
  \begin{exampleblock}{Implementação em SQL}
\begin{lstlisting}
CREATE TABLE Pedidos (
  id_pedido INT PRIMARY KEY,
  id_cliente INT,
  FOREIGN KEY (id_cliente) REFERENCES Clientes(id)
  ON DELETE CASCADE
  ON UPDATE CASCADE
);
\end{lstlisting}
  \end{exampleblock}
  \vspace{0.3cm}

\section{Paginação de Dados (LIMIT e OFFSET)}

\begin{frame}{LIMIT: Definindo a Quantidade (Teoria)}
  \begin{block}{O Sinal de Parada}
    O \textbf{LIMIT} diz ao banco de dados: "Não importa quantos registros você encontre, pare de me entregar dados assim que atingir o número $X$".
  \end{block}
  \vspace{0.4cm}
  \begin{itemize}
    \item \textbf{Analogie}: É como pedir apenas as 3 primeiras fatias de uma pizza, mesmo que ela tenha 8.
    \item \textbf{Controle}: Essencial para criar listas de "Top 10" ou destaques na tela inicial de um site.
    \item \textbf{Eficiência}: Economiza memória do servidor e banda de rede.
  \end{itemize}
\end{frame}

\begin{frame}{LIMIT: O "Corte" nos Dados (Prática)}
  \centering
  \begin{tikzpicture}[font=\tiny\sffamily]
    \foreach \i in {1,...,8} {
        \node (r\i) [draw, minimum width=3cm, fill=\ifnum\i<5 structure!20 \else gray!5 \fi] at (0, -\i*0.4) {Registro \i};
    }
    
    \draw[ultra thick, red, dashed] (-2,-1.8) -- (2,-1.8) node[right] {CORTE AQUI (LIMIT 4)};
    \node [right=1.8cm of r2, text width=3cm, font=\footnotesize, structure] {$\leftarrow$ Estes são entregues.};
    \node [right=1.8cm of r7, text width=3cm, font=\footnotesize, gray] {$\leftarrow$ Estes são ignorados.};
  \end{tikzpicture}
\end{frame}

\begin{frame}{OFFSET: Definindo o Ponto de Partida (Teoria)}
  \begin{block}{O Sinal de Salto}
    O \textbf{OFFSET} diz ao banco de dados: "Ignore os primeiros $X$ registros que você encontrar e comece a me entregar a partir do próximo".
  \end{block}
  \vspace{0.4cm}
  \begin{itemize}
    \item \textbf{Analogie}: É como começar a ler um livro a partir da página 50, ignorando as 49 anteriores.
    \item \textbf{Importância}: Sem o OFFSET, você estaria sempre preso na primeira página de resultados.
    \item \textbf{Regra}: O OFFSET sempre "pula" os dados antes de o LIMIT "cortar" os dados.
  \end{itemize}
\end{frame}

\begin{frame}{OFFSET: O "Salto" sobre os Dados (Prática)}
  \centering
  \begin{tikzpicture}[font=\tiny\sffamily]
    \foreach \i in {1,...,8} {
        \node (r\i) [draw, minimum width=3cm, fill=\ifnum\i<4 red!10 \else structure!10 \fi] at (0, -\i*0.4) {Registro \i};
    }
    
    \draw[->, >=stealth, ultra thick, red, bend left=45] (-1.8, -0.2) to node[midway, left, font=\bfseries] {SALTO (OFFSET 3)} (-1.8, -1.4);
    
    \node [right=1.8cm of r2, text width=3cm, font=\footnotesize, red] {$\leftarrow$ Ignorados (Pulados).};
    \node [right=1.8cm of r6, text width=3cm, font=\footnotesize, structure] {$\leftarrow$ Entrega começa aqui.};
  \end{tikzpicture}
\end{frame}

\begin{frame}{A Matemática da Paginação (Teoria)}
  \begin{block}{Como calcular as páginas?}
    Para implementar uma paginação (como no Google ou em uma loja virtual), usamos uma fórmula simples para calcular os valores de cada cláusula:
  \end{block}
  \vspace{0.4cm}
  \centering
  \large
  \textbf{LIMIT} = Itens por página \\
  \textbf{OFFSET} = (Página Desejada - 1) $\times$ Itens por página
  
  \vspace{0.6cm}
  \small
  \begin{itemize}
    \item \textbf{Página 1}: $(1-1) \times 10 = \mathbf{OFFSET\ 0}$
    \item \textbf{Página 2}: $(2-1) \times 10 = \mathbf{OFFSET\ 10}$
    \item \textbf{Página 3}: $(3-1) \times 10 = \mathbf{OFFSET\ 20}$
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Paginação na Vida Real (SQL)}
  \small
  Imagine um sistema com 10 produtos por página:
  \vspace{0.3cm}
  \begin{exampleblock}{Consultas para cada Página}
\begin{lstlisting}
-- Pagina 1 (Produtos 1 ao 10)
SELECT * FROM produtos LIMIT 10 OFFSET 0;

-- Pagina 2 (Produtos 11 ao 20)
SELECT * FROM produtos LIMIT 10 OFFSET 10;

-- Pagina 3 (Produtos 21 ao 30)
SELECT * FROM produtos LIMIT 10 OFFSET 20;
\end{lstlisting}
  \end{exampleblock}
  \vspace{0.2cm}
  \begin{alertblock}{Aviso Importante}
    Sempre use \textbf{ORDER BY} com paginação! Sem ordem, o banco pode pular registros diferentes a cada consulta, bagunçando a lista.
  \end{alertblock}
\end{frame}

\begin{frame}{Resumo: Por que \textbf{ORDER BY} é Vital?}
  \begin{alertblock}{Sem Ordem = Caos}
    Sensores geram dados contínuos. Se você não usar \texttt{ORDER BY data\_hora DESC}, o banco pode devolver os dados em qualquer ordem:
  \end{alertblock}
  \vspace{0.4cm}
  \begin{itemize}
    \item Você pode ver uma leitura de ontem na Página 1 e a mesma leitura na Página 2.
    \item Registros podem "sumir" entre as páginas.
    \item \textbf{Conclusão}: Paginação sem ordenação explícita é um erro grave de desenvolvimento.
  \end{itemize}

\begin{lstlisting}
SELECT DISTINCT c.cidade, AVG(p.valor) as media
FROM clientes c
JOIN pedidos p ON c.id = p.id_cliente
WHERE p.data > '2023-01-01'
GROUP BY c.cidade
HAVING AVG(p.valor) > 500
ORDER BY media DESC
LIMIT 10 OFFSET 0;
\end{lstlisting}
  \end{exampleblock}
  \vspace{0.2cm}
  \begin{itemize}
    \item \textbf{Explicação}: Busca as cidades com média de pedidos acima de 500 reais, feitas este ano, ordenando pelas cidades que mais gastam, mostrando os top 10.
  \end{itemize}
