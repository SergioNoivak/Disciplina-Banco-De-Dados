\begin{blockbox}{Configuração Importante: Google Colab e SQLite}
Para resolver este roteiro prático utilizando SQL no \textbf{Google Colab}, certifique-se de executar as etapas iniciais de instalação da extensão SQL e, \textbf{fundamentalmente}, habilitar a integridade referencial de chaves estrangeiras executando o seguinte comando no início do seu notebook:
\begin{center}
    \large\texttt{\%sql PRAGMA foreign\_keys = ON;}
\end{center}
Se essa diretiva não for executada, o SQLite aceitará inserções consistentes de chaves órfãs, prejudicando o teste das restrições DDL formuladas.
\end{blockbox}

\section{Passo 2: Modelagem das Tabelas}
Crie as tabelas base:
\begin{lstlisting}[language=SQL]
CREATE TABLE Cursos(
  id INTEGER PRIMARY KEY,
  nome TEXT NOT NULL
);

CREATE TABLE Alunos(
  id INTEGER PRIMARY KEY,
  nome TEXT NOT NULL,
  idade INTEGER NOT NULL,
  curso INTEGER,
  FOREIGN KEY (curso) REFERENCES Cursos(id)
);
\end{lstlisting}

\section{Passo 4: Praticando Window Functions}
\subsection{Agregacao Comum vs Window Function}
\begin{lstlisting}[language=SQL]
-- Agrupamento com GROUP BY
SELECT curso, AVG(idade) FROM Alunos GROUP BY curso;
\end{lstlisting}
\begin{lstlisting}[language=SQL]
-- Window Function preservando linhas
SELECT nome, curso, AVG(idade) OVER (PARTITION BY curso) AS media_curso
FROM Alunos ORDER BY curso;
\end{lstlisting}

\subsection{Acumulando Valores (Running Total)}
\begin{lstlisting}[language=SQL]
SELECT nome, curso,
       SUM(idade) OVER (PARTITION BY curso ORDER BY nome DESC) AS somatorio
FROM Alunos ORDER BY nome DESC;
\end{lstlisting}

\section{Passo 5: Praticando Views}
Crie uma view que une alunos e cursos:
\begin{lstlisting}[language=SQL]
CREATE VIEW alunos_cursos AS
SELECT Alunos.id, Alunos.nome, Cursos.nome AS curso
FROM Alunos INNER JOIN Cursos ON Alunos.curso = Cursos.id;
\end{lstlisting}
Consultando a view:
\begin{lstlisting}[language=SQL]
SELECT * FROM alunos_cursos;
\end{lstlisting}

\section{Passo 6: Praticando Triggers e Auditoria}
\subsection{Tabela de Log}
\begin{lstlisting}[language=SQL]
CREATE TABLE log_alunos_cursos (
  log_id INTEGER PRIMARY KEY AUTOINCREMENT,
  aluno_id INTEGER NOT NULL,
  aluno_nome TEXT NOT NULL,
  curso_antigo_id INTEGER,
  curso_novo_id INTEGER,
  operation_type TEXT,
  data_modificacao DATETIME DEFAULT CURRENT_TIMESTAMP
);
\end{lstlisting}
\subsection{Triggers}
\begin{lstlisting}[language=SQL]
-- UPDATE trigger
CREATE TRIGGER log_de_mudanca_cursos
AFTER UPDATE OF curso ON Alunos
FOR EACH ROW WHEN OLD.curso IS NOT NEW.curso
BEGIN
  INSERT INTO log_alunos_cursos (aluno_id, aluno_nome, curso_antigo_id, curso_novo_id, operation_type)
  VALUES (OLD.id, OLD.nome, OLD.curso, NEW.curso, 'UPDATE');
END;

-- INSERT trigger
CREATE TRIGGER log_de_insercao_alunos
AFTER INSERT ON Alunos
FOR EACH ROW
BEGIN
  INSERT INTO log_alunos_cursos (aluno_id, aluno_nome, curso_novo_id, operation_type)
  VALUES (NEW.id, NEW.nome, NEW.curso, 'INSERT');
END;

-- DELETE trigger
CREATE TRIGGER log_de_remocao_alunos
AFTER DELETE ON Alunos
FOR EACH ROW
BEGIN
  INSERT INTO log_alunos_cursos (aluno_id, aluno_nome, curso_antigo_id, operation_type)
  VALUES (OLD.id, OLD.nome, OLD.curso, 'DELETE');
END;
\end{lstlisting}
\subsection{Teste dos Triggers}
\begin{lstlisting}[language=SQL]
