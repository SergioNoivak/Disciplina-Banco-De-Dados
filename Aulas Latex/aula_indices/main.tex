\section{Revisão: Complexidade Computacional}

\begin{frame}[fragile]{Revisão 1: O que é Complexidade Computacional?}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Uma forma matemática de analisar como um algoritmo se comporta conforme a quantidade de dados ($N$) aumenta.
                \item Usamos a \textbf{Notação Big-O} ($O()$) para descrever o limite superior.
                \vspace{0.3cm}
                \item Não mede o tempo exato em segundos, mas o \textbf{crescimento das operações}.
                \item Em BD, traduz-se no número de \textbf{acessos ao disco}.
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.85, every node/.style={transform shape}]
                    \node[draw, thick, fill=structure!10, rounded corners, minimum height=1cm, minimum width=2.5cm, align=center] (dados) at (0,3) {Entrada de Dados \\ ($N$)};
                    \node[draw, thick, fill=yellow!20, circle, minimum size=1.5cm, align=center] (algo) at (0,1.2) {Algoritmo \\ $O(f(N))$};
                    \node[draw, thick, fill=red!10, rounded corners, minimum height=1cm, minimum width=2.5cm, align=center] (tempo) at (0,-0.6) {Custo \\ (Tempo/Disco)};
                    
                    \draw[->, ultra thick, structure] (dados) -- (algo);
                    \draw[->, ultra thick, red!80!black] (algo) -- (tempo);
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Revisão 2: Por que isso importa em BD?}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Bancos de dados relacionais lidam com \textbf{milhões} ou \textbf{bilhões} de registros.
                \item Um algoritmo que parece rápido com 100 linhas pode travar o sistema com 1.000.000 de linhas.
                \vspace{0.3cm}
                \item Criar um \textbf{Índice} é trocar um algoritmo ineficiente por um super otimizado!
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.9, every node/.style={transform shape}]
                    \node[align=center, draw, thick, rounded corners, fill=green!10, text width=2.5cm] (db1) at (0,2) {\textbf{Tabela Pequena} \\ 100 linhas};
                    \node[align=center, right=0.8cm of db1] (t1) {\textcolor{green!60!black}{\textbf{1 ms}} \\ \footnotesize Instantâneo};
                    
                    \node[align=center, draw, thick, rounded corners, fill=red!10, text width=2.5cm] (db2) at (0,0) {\textbf{Tabela Gigante} \\ 10.000.000 linhas};
                    \node[align=center, right=0.8cm of db2] (t2) {\textcolor{red}{\textbf{Timeout}} \\ \footnotesize Travamento!};
                    
                    \draw[->, thick, dashed] (db1) -- (t1);
                    \draw[->, thick, dashed] (db2) -- (t2);
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Revisão 3: $O(1)$ - Complexidade Constante}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item O tempo de execução é \textbf{sempre o mesmo}, não importa o volume de dados.
                \item \textbf{Cenário:} O "Santo Graal". Acesso direto via endereço físico ou Hash.
            \end{itemize}
            \vspace{0.2cm}
            \textbf{Consulta $O(1)$:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
SELECT * FROM clientes 
WHERE rowid = 42;
            \end{lstlisting}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.75, every node/.style={transform shape}]
                    \foreach \x in {1,...,5} {
                        \draw[thick, fill=gray!10] (\x*1.2, 0) rectangle ++(1, 0.8);
                    }
                    \draw[thick, fill=structure!20] (3*1.2, 0) rectangle ++(1, 0.8) node[midway] {42};
                    
                    \node[font=\bfseries, text=structure] (busca) at (3*1.2 + 0.5, 2.5) {Acesso Direto $O(1)$};
                    \draw[->, ultra thick, structure] (busca) -- (3*1.2 + 0.5, 0.8);
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Revisão 4: $O(\log N)$ - Complexidade Logarítmica}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item O tamanho do problema reduz pela metade a cada passo.
                \item \textbf{Cenário:} Altamente escalável.
                \item \textbf{Exemplo em BD:} Busca B-Tree!
            \end{itemize}
            \vspace{0.2cm}
            \textbf{Consulta $O(\log N)$:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- 'cpf' possui um índice B-Tree
SELECT * FROM clientes 
WHERE cpf = '123.456';
            \end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.9, every node/.style={transform shape}]
                    \node[draw, circle, thick, fill=structure!20] (r) at (0,2) {50};

                    \node[draw, circle, thick, fill=gray!10] (n2) at (1.5,0.5) {75};
                    \node[draw, circle, thick, fill=gray!10] (n3) at (-2.5,-1) {10};
                    \node[draw, circle, thick, fill=green!20] (n4) at (-0.5,-1) {30};

                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Revisão 5: $O(N)$ - Complexidade Linear}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item O tempo cresce proporcionalmente.
                \item \textbf{Cenário:} Péssimo para Big Data.
                \item \textbf{Exemplo em BD:} Busca sem índice (\textit{Full Table Scan}). Lê todas as linhas do disco.
            \end{itemize}
            \vspace{0.2cm}
            \textbf{Consulta $O(N)$:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- Coluna sem índice!
SELECT * FROM clientes 
WHERE nome LIKE '%Silva%';
            \end{lstlisting}
        \end{column}
        \begin{column}{0.52\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.6, every node/.style={transform shape}]
                    \foreach \x in {1,...,6} {
                        \draw[thick, fill=gray!10] (\x*1.2, 0) rectangle ++(1, 0.8);
                    }
                    \draw[thick, fill=structure!20] (6*1.2, 0) rectangle ++(1, 0.8) node[midway] {Alvo};
                    
                    \node[font=\bfseries, text=red] (busca) at (1*1.2 + 0.5, 2.5) {Busca Linear};
                    \draw[->, thick, red] (busca) -- (1*1.2 + 0.5, 0.8);
                    \draw[->, thick, red] (1*1.2 + 0.5, -0.2) to[bend right=45] (2*1.2 + 0.5, -0.2);

                    \draw[->, thick, red] (3*1.2 + 0.5, -0.2) to[bend right=45] (4*1.2 + 0.5, -0.2);

                \vspace{0.2cm} \\
                \footnotesize Lê bloco por bloco até o final!
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Revisão 6: $O(N \log N)$ - Linearítmica}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Executa uma operação $\log N$ para cada um dos $N$ itens.
                \item \textbf{Cenário:} Pior que Linear, mas é o melhor para \textbf{ordenar dados}.
                \item \textbf{Exemplo em BD:} Ordenar na memória colunas sem índice ou criar um índice novo.
            \end{itemize}
            \vspace{0.2cm}
            \textbf{Consulta $O(N \log N)$:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
SELECT * FROM clientes 
ORDER BY data_cadastro DESC;
            \end{lstlisting}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.9, every node/.style={transform shape}]
                    \node[draw, thick, fill=gray!10, minimum width=3cm] (unsorted) at (0,2) {Dados Desordenados};
                    
                    \node[draw, thick, fill=yellow!20] (m1) at (-1,0.5) {Sort};
                    \node[draw, thick, fill=yellow!20] (m2) at (1,0.5) {Sort};
                    
                    \node[draw, thick, fill=structure!20, minimum width=3cm] (sorted) at (0,-1) {Dados Ordenados};
                    
                    \draw[->, thick, dashed] (unsorted) -- (m1);
                    \draw[->, thick, dashed] (unsorted) -- (m2);
                    \draw[->, thick, structure] (m1) -- (sorted);
                    \draw[->, thick, structure] (m2) -- (sorted);
                    
                    \node[right=0.2cm of sorted, font=\footnotesize, text=red] {Custo em RAM!};
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Revisão 7: $O(N^2)$ - Complexidade Quadrática}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item O tempo cresce ao quadrado. $1000$ linhas = $1.000.000$ de operações.
                \item \textbf{Cenário:} Inaceitável para BD.
                \item \textbf{Exemplo em BD:} \textit{Nested Loop Join} sem índices. Para cada linha da Tab A, varre toda a Tab B!
            \end{itemize}
            \vspace{0.2cm}
            \textbf{Consulta $O(N^2)$:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
SELECT * FROM t1 
JOIN t2 ON t1.fk = t2.id;
-- Sem índice em fk e id!
            \end{lstlisting}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.9, every node/.style={transform shape}]
                    % Tabela A
                    \node[font=\bfseries] at (-2, 2.5) {Tab A};
                    \foreach \y in {0,1,2} {
                        \draw[thick, fill=structure!20] (-2, \y*0.8) rectangle ++(1, 0.6) node[midway] (a\y) {};
                    }
                    
                    % Tabela B
                    \node[font=\bfseries] at (2, 2.5) {Tab B};
                    \foreach \y in {-1,0,1,2,3} {
                        \draw[thick, fill=gray!20] (2, \y*0.8) rectangle ++(1, 0.6) node[midway] (b\y) {};
                    }
                    
                    % Loop Setas
                    \draw[->, thick, red] (a2.east) -- (b3.west);
                    \draw[->, thick, red] (a2.east) -- (b2.west);
                    \draw[->, thick, red] (a2.east) -- (b1.west);
                    \draw[->, thick, red] (a2.east) -- (b0.west);
                    \draw[->, thick, red] (a2.east) -- (b-1.west);
                    
                    \node[font=\footnotesize, text=red, align=center] at (0, -2) {Escaneia Tab B \\ inteira para CADA \\ linha da Tab A!};
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Revisão 8: Resumo Comparativo em BD}
    \textbf{Imagine buscar um registro específico em uma tabela com 1 Milhão de linhas:}
    
    \vspace{0.5cm}
    \begin{center}
    \renewcommand{\arraystretch}{1.5}
    \begin{tabular}{|l|l|l|}
        \hline
        \textbf{Complexidade} & \textbf{Cenário (BD)} & \textbf{Operações Estimadas} \\
        \hline
        $O(1)$ & Chave Hash (Ideal) & $1$ \\
        $O(\log N)$ & \textbf{Busca com Índice (B-Tree)} & \textbf{$\sim 20$} \\
        $O(N)$ & Busca sem Índice (Full Scan) & $1.000.000$ \\
        $O(N^2)$ & JOIN sem índice (Nested Loop) & $1.000.000.000.000$ \\
        \hline
    \end{tabular}
    \end{center}
    
    \vspace{0.3cm}
    \textit{Conclusão:} Índices $O(\log N)$ são a diferença técnica entre um sistema instantâneo e um sistema fora do ar!
\end{frame}

\begin{frame}{A Busca Sem Índice}
    Como o banco procura dados em um conjunto desordenado?
    
    \begin{alertblock}{Busca Linear (Full Table Scan)}
        A \textbf{busca sem índice percorre todas as linhas} da tabela sequencialmente, do início ao fim, verificando se cada linha atende à condição da consulta.
    \end{alertblock}

    \vspace{0.2cm}
    \begin{center}
        \begin{tikzpicture}[
            box/.style={draw, rectangle, minimum width=1cm, minimum height=0.6cm, align=center},
            >=stealth, scale=0.8, every node/.style={transform shape}
        ]
        
        \node[box, fill=gray!10] (b1) at (0,0) {Z};
        \node[box, fill=gray!10] (b2) at (1.2,0) {A};
        \node[box, fill=gray!10] (b3) at (2.4,0) {X};

\begin{frame}{Conclusão da Introdução}
    \begin{block}{O Papel dos Índices}
        Da mesma forma que as analogias anteriores, \textbf{índices aceleram consultas} no banco de dados.
    \end{block}

        \item O Índice aponta exatamente onde o registro real está no disco!
    \end{itemize}
\end{frame}

\section{Estrutura B-Tree}



\begin{frame}[fragile]{A Estrutura Padrão do SQLite}
    No \textbf{SQLite} (e em outros bancos relacionais), os índices usam a estrutura \textbf{B-Tree} (\textit{Árvore B}).
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Importante:} Não é uma Árvore Binária comum nem Rubro-Negra. A Árvore B é larga e rasa, projetada especificamente para minimizar acessos a blocos no disco.
        \vspace{0.2cm}
        \item \textbf{Árvore Balanceada:} A distância da raiz até qualquer folha é sempre a mesma. Isso garante um tempo de busca previsível e muito rápido ($O(\log N)$).
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Explicação Conceitual: Composição}
    \begin{columns}

            Uma B-Tree possui:
            \vspace{0.2cm}

                \vspace{0.2cm}
                \item \textbf{Folhas (Leaves):} Nível mais baixo. Contêm os apontamentos reais para os dados no disco.
            \end{itemize}
        \end{column}

\begin{frame}{Reorganização Automática}
    O que acontece ao fazer \texttt{INSERT} ou \texttt{DELETE}?
    
    \vspace{0.3cm}
    \begin{block}{Manutenção do Balanceamento}
        A B-Tree se \textbf{reorganiza automaticamente} para garantir que continue perfeitamente balanceada e eficiente.
    \end{block}
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Split (Divisão):} Quando um nó fica cheio (atinge o limite de tamanho do bloco no SQLite), ele é dividido em dois e a árvore cresce para cima se necessário.
        \item \textbf{Merge (Fusão):} Quando muitos registros são apagados, os nós quase vazios são mesclados para economizar espaço de armazenamento e manter as rotas rápidas.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Operações Eficientes na B-Tree}
    A B-Tree é versátil. Ela é \textbf{otimizada para} resolver com extrema rapidez as seguintes operações no \textbf{SQLite}:
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Igualdade (\texttt{=}):} Encontrar um único registro exato usando as chaves dos nós.
        \item \textbf{Intervalos (\texttt{<}, \texttt{>}, \texttt{BETWEEN}):} Ao achar o início, a estrutura permite caminhar lateralmente pelas folhas que já estão ordenadas.
        \item \textbf{Ordenação (\texttt{ORDER BY}):} Os dados na árvore já estão ordenados! O banco resolve consultas sem a necessidade de gastar memória extra.
    \end{itemize}
    
    \vspace{0.3cm}
    \begin{alertblock}{Operadores Otimizados pelo Índice B-Tree}
        \centering
        \Large \texttt{=} \qquad \texttt{<} \qquad \texttt{>} \qquad \texttt{BETWEEN} \qquad \texttt{ORDER BY}
    \end{alertblock}
\end{frame}

\section{Analisando Consultas}

\begin{frame}[fragile]{O comando EXPLAIN QUERY PLAN}
    \begin{itemize}
        \item Como ter certeza de que o banco de dados está realmente usando o índice que criamos?
        \item O SQLite fornece o comando \textbf{EXPLAIN QUERY PLAN}.
        \item Ele não executa a consulta, apenas \textquotedblleft lê a mente\textquotedblright{} do otimizador do banco e nos diz \textbf{como} ele planeja buscar os dados no disco.
    \end{itemize}
    
    \vspace{0.3cm}
    \begin{block}{Sintaxe}
        Basta adicionar as palavras antes de qualquer \texttt{SELECT}, \texttt{UPDATE} ou \texttt{DELETE}:
        \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
EXPLAIN QUERY PLAN
SELECT * FROM clientes WHERE email = 'joao@email.com';
        \end{lstlisting}
    \end{block}
\end{frame}

\begin{frame}{Lendo o Resultado: SCAN vs SEARCH}
    O \texttt{EXPLAIN QUERY PLAN} devolve uma mensagem indicando o método de acesso escolhido pelo SQLite. Os dois mais comuns são:
    
    \vspace{0.3cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{SCAN TABLE}
                \textbf{O que significa:} Busca Linear $O(N)$.
                \begin{itemize}
                    \item O banco vai ler \textbf{todas} as linhas da tabela.
                    \item Sinal de alerta vermelho! Faltam índices ou a consulta está mal escrita.
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{exampleblock}{SEARCH TABLE}
                \textbf{O que significa:} Busca Logarítmica $O(\log N)$.
                \begin{itemize}
                    \item O banco vai usar um \textbf{Índice} (Árvore B).
                    \item Caminho feliz! A consulta será incrivelmente rápida e eficiente.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Exemplo Prático: Tabela Sem Índice (SCAN)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Consulta:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
EXPLAIN QUERY PLAN
SELECT * FROM clientes 
WHERE telefone = '555-1234';
            \end{lstlisting}
            \vspace{0.2cm}
            \textbf{Saída do SQLite:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize, backgroundcolor=\color{red!10}]
SCAN TABLE clientes
            \end{lstlisting}
            \vspace{0.2cm}
            \textbf{Veredito:} O banco fará um \textit{Full Table Scan} $O(N)$. Desempenho péssimo para tabelas grandes.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.7, every node/.style={transform shape}]
                    \node[draw, thick, fill=red!10, text width=3cm, align=center, rounded corners, minimum height=1cm] (scan) at (0, 0) {\textbf{SCAN TABLE} \\ \footnotesize Leitura Sequencial};
                    \node[font=\Huge, text=red] (lupa) at (0, 1.5) {$\bigotimes$};
                    \draw[->, thick, red] (lupa) -- (scan);
                    
                    \foreach \x in {-2,-1,0,1,2} {
                        \draw[thick, fill=gray!20] (\x*1.2, -1.5) rectangle ++(1, 0.5);
                        \draw[->, red] (\x*1.2+0.5, -0.3) -- (\x*1.2+0.5, -1);
                    }
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Exemplo Prático: Com Índice (SEARCH)}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Criando o Índice e Consultando:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
CREATE INDEX idx_tel 
ON clientes(telefone);

EXPLAIN QUERY PLAN
SELECT * FROM clientes 
WHERE telefone = '555-1234';
            \end{lstlisting}
            \vspace{0.2cm}
            \textbf{Saída do SQLite:}
            \begin{lstlisting}[basicstyle=\ttfamily\scriptsize, backgroundcolor=\color{green!10}]
SEARCH TABLE clientes 
USING INDEX idx_tel (telefone=?)
            \end{lstlisting}
            \vspace{0.2cm}
            \textbf{Veredito:} O banco agora vai usar a Árvore B $O(\log N)$ do índice criado para pular direto para o dado!
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}

\begin{frame}{PostgreSQL 1: A Extensibilidade de Índices}
    O \textbf{PostgreSQL} é mundialmente famoso pela sua extensibilidade e suporte nativo a múltiplos métodos de acesso a índices:
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{B-Tree:} O padrão confiável (para dados escalares comuns e comparações de ordenação).
        \item \textbf{Hash:} Totalmente focado em buscas exatas de igualdade ($O(1)$).
        \item \textbf{Expression \& Partial:} Índices customizados para subconjuntos de dados ou cálculos específicos.
        \item \textbf{GIN (Generalized Inverted Index):} Para registros que contêm múltiplos valores internos.
        \item \textbf{GiST (Generalized Search Tree):} Para estruturas geométricas, espaciais, hierárquicas e faixas.
        \item \textbf{BRIN (Block Range Index):} Para tabelas gigantes ordenadas fisicamente (Big Data).
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{PostgreSQL 2: Índices HASH - Busca $O(1)$}
    Diferente de bancos mais simples, o PostgreSQL possui suporte robusto a índices do tipo \textbf{Hash}:
    
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Como Funciona:} Aplica uma função de espalhamento sobre a coluna. Mapeia diretamente o valor para o endereço do bloco físico ($O(1)$).
        \item \textbf{Limitação Importante:} Só funciona com o operador de igualdade (\texttt{=}). Não serve para operadores de comparação (\texttt{<}, \texttt{>}), intervalos ou ordenações.
        \item \textbf{Segurança (WAL):} Desde a versão 10, índices Hash são 100\% seguros contra quedas de energia (WAL-logged) e replicáveis.
    \end{itemize}
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- Criando um indice HASH
CREATE INDEX idx_hash_token 
ON usuarios USING HASH (token_acesso);
    \end{lstlisting}
\end{frame}

\begin{frame}{PostgreSQL 3: HASH - Funcionamento Interno}
    \begin{columns}

            \begin{itemize}
                \item O valor do campo é transformado em um número inteiro (Hash).
                \item Esse hash aponta diretamente para um \textbf{Bucket} (balde).
                \item O balde contém o ponteiro direto da linha física (TID) no disco.
                \item Evita percorrer qualquer árvore, gerando velocidade instantânea!
            \end{itemize}
        \end{column}

            \begin{center}

\end{frame}

\begin{frame}[fragile]{PostgreSQL 4: Índices de Expressão}
    Por padrão, os índices olham apenas o valor bruto da coluna. O que acontece se filtrarmos usando funções?
    
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{O Problema:} Uma busca usando \texttt{LOWER(email)} ignora um índice normal criado na coluna \texttt{email}!
        \item \textbf{A Solução:} Indexar o resultado da própria função aplicada à coluna.
    \end{itemize}
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- Criando indice de Expressao
CREATE INDEX idx_email_lower 
ON clientes ( LOWER(email) );

-- Query super rapida O(log N) usando o indice:
SELECT * FROM clientes 
WHERE LOWER(email) = 'ana@email.com';
    \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{PostgreSQL 5: Índices Parciais}
    Por que gastar disco indexando linhas que raramente ou nunca serão consultadas?
    
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Conceito:} Um índice parcial contém apenas as referências para um subconjunto dos dados que atende a uma condição lógica (\texttt{WHERE}).
        \item \textbf{Vantagens:} Reduz drasticamente o tamanho do índice no disco e acelera as atualizações de escrita (\texttt{INSERT} / \texttt{UPDATE}).
    \end{itemize}
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- Indexando apenas usuarios ativos
CREATE INDEX idx_usuarios_ativos 
ON usuarios (email) 
WHERE ativo = TRUE;

-- Otimiza perfeitamente queries com o mesmo filtro:
SELECT * FROM usuarios 
WHERE email = 'admin@sistema.com' AND ativo = TRUE;
    \end{lstlisting}
\end{frame}

\begin{frame}{PostgreSQL 6: Índices Parciais - Economia de Recursos}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item \textbf{Sem Índice Parcial:}
                O índice B-Tree guarda ponteiros para todos os 10 milhões de clientes, incluindo inativos e excluídos.
                \vspace{0.2cm}
                \item \textbf{Com Índice Parcial:}
                Indexamos apenas os 1\% de clientes ativos.
                \item Redução enorme em I/O e uso de RAM!
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}

                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{PostgreSQL 7: O que é o GIN?}
    O \textbf{GIN} (\textit{Generalized Inverted Index}) é projetado para lidar com casos em que \textbf{um único registro contém múltiplos valores internos}.
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Onde B-Tree Falha:} Uma árvore B assume uma relação de 1 valor chave por linha. Mas e se tivermos uma coluna com uma lista (Array) de tags ou um objeto JSON?
        \item \textbf{Abordagem GIN:} O GIN 'inverte' a relação. Ele mapeia os elementos individuais internos (ex: palavras, chaves, tags) de volta para todas as linhas (TIDs) que os contêm.
        \item \textbf{Usos principais:} Arrays indexados, documentos JSONB e buscas por texto completo.
    \end{itemize}
\end{frame}

\begin{frame}{PostgreSQL 8: GIN - Estrutura Interna}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item Cada elemento interno (ex: 'SQL', 'Python') torna-se uma chave na Árvore do GIN.
                \item Abaixo de cada chave, existe uma lista (Postings List) contendo os IDs de todos os registros que contêm aquele valor.
                \item Encontrar dados em Arrays é incrivelmente rápido!
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{center}
                \begin{tikzpicture}[>=stealth, scale=0.75, every node/.style={transform shape}]
                    % Chaves GIN
                    \node[draw, thick, fill=structure!10, rounded corners] (k1) at (-1.5, 2) {'SQL'};
                    \node[draw, thick, fill=structure!10, rounded corners] (k2) at (1.5, 2) {'Python'};
                    
                    % Listas de apontamento
                    \node[draw, thick, fill=gray!20, rectangle split, rectangle split parts=3] (l1) at (-1.5, 0) {Linha 1 \nodepart{second} Linha 3 \nodepart{third} Linha 5};
                    \node[draw, thick, fill=gray!20, rectangle split, rectangle split parts=2] (l2) at (1.5, 0.25) {Linha 1 \nodepart{second} Linha 5};
                    
                    \draw[->, thick, structure] (k1) -- (l1);
                    \draw[->, thick, structure] (k2) -- (l2);
                \end{tikzpicture}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{PostgreSQL 9: GIN - Arrays Indexados}
    Suponha que cada desenvolvedor cadastrado possua uma lista (Array) de tecnologias de interesse:
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- Tabela com coluna tipo Array
CREATE TABLE desenvolvedores (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100),
    tecnologias VARCHAR(50)[] -- Lista de stacks
);

-- Criando um indice GIN
CREATE INDEX idx_dev_techs 
ON desenvolvedores USING GIN (tecnologias);

-- Query usando o operador de contencao (@>)
SELECT * FROM desenvolvedores 
WHERE tecnologias @> ARRAY['PostgreSQL', 'Python'];
    \end{lstlisting}
    
    \vspace{0.1cm}
    \textit{Nota:} Sem o índice GIN, a busca com operador `@>` exigiria ler e varrer linearmente cada linha da tabela!
\end{frame}

\begin{frame}[fragile]{PostgreSQL 10: GIN - Documentos JSONB}
    O tipo \texttt{JSONB} armazena objetos JSON binários. Combinado com o GIN, entrega performance excelente para consultas semi-estruturadas:
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
CREATE TABLE produtos (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100),
    atributos JSONB
);

-- Indexando todas as chaves e valores internos do JSONB
CREATE INDEX idx_prod_atributos ON produtos USING GIN (atributos);

-- Busca direta no JSON com velocidade incrivel
SELECT * FROM produtos 
WHERE atributos @> '{"categoria": "Eletro", "voltagem": "220v"}';
    \end{lstlisting}
    
    \vspace{0.15cm}
    \begin{itemize}
        \item \textbf{jsonb\_ops (padrão):} Indexa todas as chaves, valores e subchaves do documento JSON.
        \item \textbf{jsonb\_path\_ops:} Ocupa menos espaço físico, porém focado em caminhos de navegação completos.
    \end{itemize}
\end{frame}

\begin{frame}{PostgreSQL 11: O que é o GiST?}
    A árvore \textbf{GiST} (\textit{Generalized Search Tree}) é um \textbf{template estrutural} altamente flexível.
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{Customização Total:} Permite definir novas regras de indexação para tipos de dados complexos que precisam ser buscados de forma balanceada.
        \item \textbf{Como Funciona:} Ao contrário de uma B-Tree tradicional que divide dados rigidamente de forma linear, a árvore GiST agrupa dados baseado em predicados de exclusão mútua (ex: 'contido em', 'sobreposto com').
        \item \textbf{Usos típicos:} Dados geográficos (coordenadas), intervalos de datas ou numéricos e buscas textuais especiais.
    \end{itemize}
\end{frame}

\begin{frame}{PostgreSQL 12: GiST - A Estrutura Hierárquica}
    A árvore do GiST é estruturada em torno de caixas delimitadoras ou condições que podem se sobrepor:
    
    \vspace{0.3cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item \textbf{Nós Superiores:} Guardam predicados gerais abrangentes (ex: 'Região Centro-Oeste').
                \item \textbf{Nós Inferiores:} Refinam a busca para sub-regiões (ex: 'Goiás' -> 'Cidade de Rio Verde').
                \item Isso permite descartar ramos inteiros de busca no disco instantaneamente!
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}

\begin{frame}[fragile]{PostgreSQL 13: Índices Espaciais (PostGIS)}
    O ecossistema de processamento espacial do PostgreSQL (\textbf{PostGIS}) depende fundamentalmente dos índices do tipo GiST:
    
    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Desafio:} Coordenadas no mapa (Latitude, Longitude) não possuem ordem linear. Como indexar duas dimensões geográficas?
        \item \textbf{Solução:} O GiST agrupa dados geométricos usando \textbf{Bounding Boxes} (Retângulos Mínimos Envolventes).
    \end{itemize}
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- Habilitando extensao espacial no PostgreSQL
CREATE EXTENSION postgis;

-- Tabela com dados geometricos
CREATE TABLE filiais (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100),
    localizacao GEOMETRY(Point, 4326) -- Tipo Ponto Geografico
);

-- Indispensavel para consultas geometricas espaciais rapidas!
CREATE INDEX idx_filiais_geo ON filiais USING GIST (localizacao);
    \end{lstlisting}
\end{frame}

\begin{frame}{PostgreSQL 15: O que são Trigramas?}
    O módulo \textbf{pg\_trgm} introduz uma técnica poderosa baseada em \textbf{Trigramas} para resolver buscas parciais textuais flexíveis.
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{A Limitação do B-Tree:} Um índice comum B-Tree só serve para buscas com curinga no final (\texttt{LIKE 'Silva\%'}). Se usarmos \texttt{LIKE '\%Silva\%'}, o índice B-Tree é ignorado.
        \item \textbf{O Conceito do Trigrama:} Qualquer texto é quebrado em pequenas fatias sobrepostas de 3 letras consecutivas.
        \item \textbf{Exemplo:} A palavra 'BANCO' gera os trigramas:
        \begin{center}
            \texttt{'  b'}, \texttt{' ba'}, \texttt{'ban'}, \texttt{'anc'}, \texttt{'nco'}, \texttt{'co '}
        \end{center}
        \item \textbf{Poder Máximo:} Permite buscas ultra rápidas mesmo com termos no início e fim (\texttt{\%termo\%}) e até correção ortográfica (Fuzzy Match)!
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{PostgreSQL 16: Trigramas em Ação com GIN}
    Instalando a extensão e criando um índice que armazena os trigramas das strings:
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
-- Ativando o modulo oficial de trigramas
CREATE EXTENSION pg_trgm;

CREATE TABLE usuarios (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100)
);

-- Criando indice GIN usando a classe de operador de trigramas
CREATE INDEX idx_usuarios_nome_trgm 
ON usuarios USING GIN (nome gin_trgm_ops);

-- Consulta O(log N) altamente otimizada, mesmo com % no inicio!
SELECT * FROM usuarios 
WHERE nome LIKE '%Sergio%';
    \end{lstlisting}
\end{frame}

\begin{frame}{PostgreSQL 18: Busca Textual Avançada (Full-Text Search)}
    A busca por texto completo nativa do PostgreSQL é capaz de substituir sistemas externos inteiros (como Elasticsearch) em projetos de médio porte:
    
    \vspace{0.3cm}
    \begin{itemize}
        \item \textbf{tsvector:} Tipo de dados que representa um documento analisado, aplicando radicalização (\textit{stemming}) e eliminando palavras irrelevantes (\textit{stop words} como 'e', 'de', 'um').
        \item \textbf{tsquery:} Representa a nossa consulta contendo operadores booleanos (\texttt{\&} (E), \texttt{|} (OU), \texttt{!} (NÃO)).
        \item \textbf{Exemplo:} As palavras 'correndo', 'corri', 'correu' são todas simplificadas e convertidas para o mesmo radical: \texttt{'corr'}.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{PostgreSQL 19: FTS na Prática}
    Implementando busca de artigos usando indexação por texto completo com o GIN:
    
    \vspace{0.2cm}
    \begin{lstlisting}[basicstyle=\ttfamily\scriptsize]
CREATE TABLE artigos (
    id SERIAL PRIMARY KEY,
    titulo VARCHAR(200),
    conteudo TEXT
);

-- Criando indice GIN usando a funcao to_tsvector em portugues
CREATE INDEX idx_artigos_fts 
ON artigos USING GIN (to_tsvector('portuguese', conteudo));

-- Query extremamente rapida usando o operador @@
SELECT * FROM artigos 
WHERE to_tsvector('portuguese', conteudo) @@ to_tsquery('portuguese', 'bancos & indices');
    \end{lstlisting}
    
    \vspace{0.1cm}
    \begin{itemize}
        \item A query acima busca e encontra artigos que contêm radicais de 'bancos' e 'índices', mesmo em flexões gramaticais diferentes.
    \end{itemize}
\end{frame}

\begin{frame}{PostgreSQL 20: Matriz de Decisão de Índices}
    \begin{center}
    \renewcommand{\arraystretch}{1.3}
    \scriptsize
    \begin{tabular}{|l|l|l|p{3.5cm}|}
        \hline
        \textbf{Método} & \textbf{Complexidade} & \textbf{Melhor Operador} & \textbf{Cenário Ideal (BD)} \\
        \hline
        \textbf{B-Tree} & $O(\log N)$ & \texttt{=}, \texttt{<}, \texttt{>}, \texttt{BETWEEN} & Dados comuns ordenáveis (Datas, IDs, Nomes). \\
        \hline
        \textbf{Hash} & $O(1)$ & \texttt{=} & Chaves de token, UUIDs (busca exata). \\
        \hline
        \textbf{GIN} & $O(\log N)$ & \texttt{@>}, \texttt{?}, \texttt{@@} & Documentos JSONB, Arrays, Busca Textual (FTS). \\
        \hline
        \textbf{GiST} & $O(\log N)$ & \texttt{\&\&}, \texttt{<<}, \texttt{>>} & Dados Geográficos (PostGIS), Intervalos numéricos/datas. \\
        \hline
        \textbf{BRIN} & $O(N)$ parcial & Limites de bloco & Logs gigantes ou dados históricos ordenados fisicamente. \\
        \hline
    \end{tabular}
    \end{center}
    
    \vspace{0.25cm}
    \textit{Dica de Ouro:} Use sempre o comando \texttt{EXPLAIN ANALYZE} no PostgreSQL para certificar suas escolhas de índice em ambiente de produção!
\end{frame}
