% INNER JOIN
\begin{center}
\textbf{INNER JOIN} \\[0.5em]
\begin{tikzpicture}[scale=0.6]
    \begin{scope}
        \clip (0,0) circle (1.2);
        \fill[maincolor!50] (1.2,0) circle (1.2);
    \end{scope}
    \draw (0,0) circle (1.2);
    \draw (1.2,0) circle (1.2);
    \node at (-0.5,0) {\tiny Alunos};
    \node at (1.7,0) {\tiny Cursos};
\end{tikzpicture}
\end{center}
\begin{lstlisting}
SELECT A.nome, C.nome_curso
FROM Alunos A
INNER JOIN Cursos C ON A.c_id = C.id;
\end{lstlisting}

\vspace{1em}

% LEFT EXCLUSIVE
\begin{center}
\textbf{LEFT EXCLUSIVE} \\[0.5em]
\begin{tikzpicture}[scale=0.6]
    \fill[maincolor!50] (0,0) circle (1.2);
    \begin{scope}
        \clip (1.2,0) circle (1.2);
        \fill[white] (0,0) circle (1.2);
    \end{scope}
    \draw (0,0) circle (1.2);
    \draw (1.2,0) circle (1.2);
    \node at (-0.5,0) {\tiny Alunos};
    \node at (1.7,0) {\tiny Cursos};
\end{tikzpicture}
\end{center}
\begin{lstlisting}
SELECT A.nome, C.nome_curso
FROM Alunos A
LEFT JOIN Cursos C ON A.c_id = C.id
WHERE C.id IS NULL;
\end{lstlisting}

% FULL EXCLUSIVE (Anti-Join)
\begin{center}
\textbf{FULL EXCLUSIVE (Anti-Join)} \\[0.5em]
\begin{tikzpicture}[scale=0.6]
    \fill[maincolor!50] (0,0) circle (1.2);
    \fill[maincolor!50] (1.2,0) circle (1.2);
    \begin{scope}
        \clip (0,0) circle (1.2);
        \fill[white] (1.2,0) circle (1.2);
    \end{scope}
    \draw (0,0) circle (1.2);
    \draw (1.2,0) circle (1.2);
    \node at (-0.5,0) {\tiny Alunos};
    \node at (1.7,0) {\tiny Cursos};
\end{tikzpicture}
\end{center}
\begin{lstlisting}
SELECT A.nome, C.nome_curso
FROM Alunos A
FULL JOIN Cursos C ON A.c_id = C.id
WHERE A.id IS NULL OR C.id IS NULL;
\end{lstlisting}
