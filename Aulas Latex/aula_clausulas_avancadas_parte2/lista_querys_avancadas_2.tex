        % Linha entidade -> relacionamento
        \draw[line] (colab.south) -- (supervisiona.north);

        % Loop: sai do lado direito do losango, sobe e entra pelo lado direito da entidade

            (supervisiona.east) -- ++(1.8,0) -- ++(0,2.5) -- (colab.east);
