\section{3NF: Definição e Dependência Transitiva}

\begin{frame}{3ª Forma Normal (3NF)}
  \begin{block}{Definição}
    Uma relação está na \textbf{3NF} se, e somente se:
    \begin{enumerate}
      \item Já estiver na \textbf{2ª Forma Normal (2NF)}.
      \item \textbf{Não existirem dependências transitivas} entre atributos não-chave.
    \end{enumerate}
  \end{block}
  \vspace{0.3cm}
  \begin{itemize}
    \item A 3NF visa eliminar as \textbf{Dependências Funcionais Transitivas}.
    \item Todo atributo não-chave deve depender \textbf{diretamente} da chave primária, e \textbf{somente} dela.
  \end{itemize}
  \vspace{0.3cm}
  \begin{exampleblock}{Intuição}
    Se um atributo depende de outro atributo que \textbf{não é chave}, estamos perante uma dependência transitiva --- um problema a ser eliminado.
  \end{exampleblock}
\end{frame}

\begin{frame}{O que é Dependência Transitiva?}
  \begin{block}{Definição}
    Dados os atributos $A$, $B$ e $C$ de uma relação, existe uma \textbf{dependência transitiva} quando:
    \[
      A \to B \quad \text{e} \quad B \to C \quad \Rightarrow \quad A \to C \text{ (transitivamente)}
    \]
    sendo $B$ um atributo \textbf{não-chave}.
  \end{block}
  \vspace{0.4cm}
  \begin{itemize}
    \item $A$ é a chave primária.
    \item $B$ é um atributo intermediário (não-chave) que determina $C$.
    \item $C$ não depende diretamente de $A$, mas sim de $B$.
  \end{itemize}
  \vspace{0.3cm}
  \begin{alertblock}{Problema}
    A dependência $B \to C$ torna os dados de $C$ redundantes: se mudar o valor de $B$ em uma linha, $C$ precisa ser atualizado em todas as linhas com o mesmo $B$.
  \end{alertblock}
\end{frame}

\section{3NF: Algoritmo de Verificação}

\begin{frame}{Algoritmo: Verificação de Dependências Transitivas}
\small

% \begin{block}{Ideia}
% Dado um conjunto de dependências funcionais $F$, queremos verificar se existe:
% \[
% A \to B \quad \text{e} \quad B \to C \quad \Rightarrow \quad A \to C \text{ (transitiva)}
% \]
% onde $B$ não é chave.
% \end{block}

\vspace{0.3cm}
\begin{block}{Pseudocódigo (Detecção de Dependência Transitiva)}
\begin{algorithmic}[1]
\Procedure{ExisteDependenciaTransitiva}{$R, F, K$}
    \State \Comment{$R$: relação, $F$: dependências funcionais, $K$: chave candidata}

    \ForAll{$(X \to Y) \in F$}
        \State \Comment{Procura uma primeira dependência}

        \ForAll{$(Y \to Z) \in F$}
            \State \Comment{Procura uma segunda dependência encadeada}

            \If{$X \neq Y$ \textbf{and} $Y \neq Z$}
                \State \Comment{Evita dependências triviais ou repetidas}

                \If{$Y \not\subseteq K$}
                    \State \Comment{$Y$ NÃO é chave (condição essencial)}

                    \If{$Z \not\subseteq X$}
                        \State \Comment{$Z$ não depende diretamente de $X$}

                        \State \Comment{Encontramos: $X \to Y$ e $Y \to Z$}
                        \State \Comment{Logo: $X \to Z$ (dependência transitiva)}

                        \State \Return \textbf{true}
                    \EndIf
                \EndIf

            \EndIf

        \EndFor
    \EndFor

    \State \Comment{Nenhuma dependência transitiva encontrada}
    \State \Return \textbf{false}
\EndProcedure
\end{algorithmic}
\end{block}

\vspace{0.3cm}

\begin{alertblock}{Interpretação}
\begin{itemize}
    \item Detecta cadeias do tipo $X \to Y \to Z$
    \item Garante que $Y$ é atributo não-chave
    \item Identifica quando $Z$ depende indiretamente de $X$
\end{itemize}
\end{alertblock}

\end{frame}

\begin{frame}{Exemplo: Tabela FUNCIONARIOS (Não está na 3NF)}
  \small
  Chave Primária Simples: \textbf{ID\_Fun}
  \vspace{0.2cm}
  \centering
  \begin{tabular}{|c|l|c|l|}
    \hline
    \textbf{ID\_Fun} & \textbf{Nome} & \textbf{Cod\_Cargo} & \textbf{Salario\_Cargo} \\ \hline
    101 & Alice   & C1 & \color{red}R\$ 5.000 \\ \hline
    102 & Bob     & C2 & \color{red}R\$ 8.000 \\ \hline
    103 & Carol   & C1 & \color{red}R\$ 5.000 \\ \hline
    104 & David   & C2 & \color{red}R\$ 8.000 \\ \hline
  \end{tabular}
\end{frame}

\begin{frame}{Execução do Algoritmo (Com Dependência Transitiva)}
\small

\begin{block}{Entrada}
\begin{itemize}
    \item $F = \{\, \text{ID\_Fun} \to \text{Cod\_Cargo},\ \text{Cod\_Cargo} \to \text{Salario} \,\}$
    \item $K = \{\text{ID\_Fun}\}$
\end{itemize}
\end{block}

\vspace{0.3cm}

\begin{block}{Execução}
\begin{enumerate}
    \item Seleciona $\text{ID\_Fun} \to \text{Cod\_Cargo}$
    \item Seleciona $\text{Cod\_Cargo} \to \text{Salario}$

    \item Verifica:
    \begin{itemize}
        \item $\text{Cod\_Cargo} \notin K$ \checkmark
        \item Dependência indireta detectada \checkmark
    \end{itemize}

    \item \textbf{Retorna true}
\end{enumerate}
\end{block}

\begin{alertblock}{Conclusão}
Existe:
\[
\text{ID\_Fun} \to \text{Cod\_Cargo} \to \text{Salario}
\]
Violação da 3FN.
\end{alertblock}

\end{frame}






\begin{frame}{Execução do Algoritmo (Sem Dependência Transitiva)}
\small

\begin{block}{Entrada}
\begin{itemize}
    \item $F = \{\, \text{ID\_Fun} \to \text{Nome},\ \text{ID\_Fun} \to \text{Cod\_Cargo} \,\}$
    \item $K = \{\text{ID\_Fun}\}$
\end{itemize}
\end{block}

\vspace{0.3cm}

\begin{block}{Execução}
\begin{enumerate}
    \item Avalia $\text{ID\_Fun} \to \text{Nome}$
    \item Não existe $\text{Nome} \to Z$

    \item Avalia $\text{ID\_Fun} \to \text{Cod\_Cargo}$
    \item Não existe $\text{Cod\_Cargo} \to Z$

    \item Nenhuma cadeia transitiva encontrada
\end{enumerate}
\end{block}

\begin{alertblock}{Conclusão}
\begin{itemize}
    \item Retorno: \textbf{false}
    \item Relação está na \textbf{3FN}
\end{itemize}
\end{alertblock}

\end{frame}
\begin{frame}{Exemplo: Tabela FUNCIONARIOS (Não está na 3NF)}
  \small
  Chave Primária Simples: \textbf{ID\_Fun}
  \vspace{0.2cm}
  \centering
  \begin{tabular}{|c|l|c|l|}
    \hline
    \textbf{ID\_Fun} & \textbf{Nome} & \textbf{Cod\_Cargo} & \textbf{Salario\_Cargo} \\ \hline
    101 & Alice   & C1 & \color{red}R\$ 5.000 \\ \hline
    102 & Bob     & C2 & \color{red}R\$ 8.000 \\ \hline
    103 & Carol   & C1 & \color{red}R\$ 5.000 \\ \hline
    104 & David   & C2 & \color{red}R\$ 8.000 \\ \hline
  \end{tabular}

  \vspace{0.4cm}
  \flushleft
  \textbf{Dependências identificadas:}
  \begin{itemize}
    \item \textbf{OK:} $\text{ID\_Fun} \to \text{Nome}$ e $\text{ID\_Fun} \to \text{Cod\_Cargo}$ \quad (diretas)
    \item \textbf{ERRO:} $\text{Cod\_Cargo} \to \text{Salario\_Cargo}$ \quad (transitiva!)
  \end{itemize}
  \textit{O \textbf{Cod\_Cargo} não é chave, mas determina o \textbf{Salario\_Cargo}: dependência transitiva.}
\end{frame}

  \end{tikzpicture}
\end{frame}

\begin{frame}{Solução: Decomposição para a 3NF}
\small

\begin{block}{Processo de Normalização}
Para eliminar a dependência transitiva $\text{Cod\_Cargo} \to \text{Salario\_Cargo}$, decompomos a relação em duas:
\end{block}

\vspace{0.3cm}

\begin{columns}[t]
    \begin{column}{0.48\textwidth}
        \begin{exampleblock}{R1: FUNCIONARIOS (3FN)}
        \textbf{Chave:} $\underline{\text{ID\_Fun}}$

        \centering
        \vspace{0.1cm}
        \begin{tabular}{|c|l|c|}
        \hline
        \textbf{ID\_Fun} & \textbf{Nome} & \textbf{Cod\_Cargo} \\ \hline
        101 & Alice & C1 \\ \hline
        102 & Bob   & C2 \\ \hline
        103 & Carol & C1 \\ \hline
        \end{tabular}
        \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
        \begin{exampleblock}{R2: CARGOS (3FN)}
        \textbf{Chave:} $\underline{\text{Cod\_Cargo}}$

        \centering
        \vspace{0.1cm}
        \begin{tabular}{|c|r|}
        \hline
        \textbf{Cod\_Cargo} & \textbf{Salario} \\ \hline
        C1 & R\$ 5.000 \\ \hline
        C2 & R\$ 8.000 \\ \hline
        \end{tabular}
        \end{exampleblock}
    \end{column}
\end{columns}

\vspace{0.4cm}

\begin{block}{Resultado}
\begin{itemize}
  \item O salário é armazenado \textbf{uma única vez} na tabela CARGOS.
  \item Não há mais redundância nem risco de inconsistência.
  \item \textbf{Cod\_Cargo} em FUNCIONARIOS passa a ser uma \textbf{Chave Estrangeira (FK)}.
\end{itemize}
\end{block}
\end{frame}

      {Agora o \textbf{Salario\_Cargo} aparece apenas uma vez, eliminando a redundância.};
  \end{tikzpicture}
\end{frame}

\small
\centering
\begin{tabular}{|l|l|l|}
  \hline
  \textbf{Forma Normal} & \textbf{Elimina} & \textbf{Condição Principal} \\ \hline
  1FN & Atributos multivalorados & Valores atômicos e chave primária \\ \hline
  2FN & Dependência parcial & Depender de \textbf{toda} a chave \\ \hline
  3FN & Dependência transitiva & Não depender de atributo não-chave \\ \hline
\end{tabular}

\section{3NF: Checklist e Resumo}

\begin{frame}{Checklist para 3NF}
  Para garantir que uma tabela está na \textbf{3NF}, verifique:
  \vspace{0.4cm}
  \begin{itemize}
    \item[$\square$] A tabela já está na \textbf{2NF}?
    \item[$\square$] Existe algum atributo não-chave que depende de outro atributo não-chave?
    \item[$\square$] Se mudar um valor em uma coluna não-chave, isso força a repetição de um valor em outra coluna?
    \item[$\square$] Todos os atributos dependem \textbf{diretamente} da chave primária?
  \end{itemize}
  \vspace{0.4cm}
  \begin{block}{Dica de Ouro}
    Se você percebe que a mesma informação se repete em várias linhas por causa de um campo que \textbf{não é a chave}, esse campo intermediário provavelmente cria uma dependência transitiva --- e a 3NF está sendo violada.
  \end{block}
\end{frame}

\section{Forma Normal de Boyce-Codd (BCNF)}

% SLIDE 1 — Definição da BCNF
\begin{frame}{Forma Normal de Boyce-Codd (BCNF)}
  \begin{block}{Definição}
    Uma relação está na \textbf{BCNF} se, e somente se, para toda dependência funcional não trivial
    \[
      X \to Y
    \]
    $X$ for uma \textbf{superchave} da relação.
  \end{block}
  \vspace{0.3cm}
  \begin{itemize}
    \item A BCNF é uma versão \textbf{mais restrita} da 3NF.
    \item Na 3NF, $Y$ pode ser atributo primo (parte de alguma chave candidata) mesmo que $X$ não seja superchave.
    \item Na BCNF, \textbf{não há exceção}: todo determinante deve ser superchave.
  \end{itemize}
  \vspace{0.3cm}
  \begin{alertblock}{Quando a 3NF não basta?}
    Quando existem \textbf{múltiplas chaves candidatas sobrepostas} (que compartilham atributos), a 3NF pode ainda permitir anomalias que a BCNF elimina.
  \end{alertblock}
\end{frame}

% SLIDE 2 — Comparação 3NF × BCNF
\begin{frame}{Comparação: 3NF vs BCNF}
\small
\centering
\begin{tabular}{|l|c|c|}
  \hline
  \textbf{Critério} & \textbf{3NF} & \textbf{BCNF} \\ \hline
  Elimina dep. parciais            & \checkmark & \checkmark \\ \hline
  Elimina dep. transitivas         & \checkmark & \checkmark \\ \hline
  $X$ deve ser superchave          & Às vezes   & \textbf{Sempre} \\ \hline
  $Y$ pode ser atributo primo      & \textbf{Sim} & \textbf{Não} \\ \hline
  Garante preservação de dep.      & \textbf{Sim} & Nem sempre \\ \hline
  Garante junção sem perdas        & \checkmark & \checkmark \\ \hline
  Mais restritiva                  & ---        & \textbf{Sim} \\ \hline
\end{tabular}

\vspace{0.4cm}
\begin{block}{Relação hierárquica}
\centering
\begin{tikzpicture}[node distance=0cm]
  \node (fn1) [draw, fill=blue!10, minimum width=8cm, minimum height=0.7cm, font=\small\bfseries] {1ª Forma Normal (1NF)};
  \node (fn2) [draw, fill=green!10, minimum width=6.5cm, minimum height=0.7cm, font=\small\bfseries, below=-\pgflinewidth of fn1] {2ª Forma Normal (2NF)};
  \node (fn3) [draw, fill=orange!10, minimum width=5cm, minimum height=0.7cm, font=\small\bfseries, below=-\pgflinewidth of fn2] {3ª Forma Normal (3NF)};
  \node (bcnf)[draw, fill=red!10,    minimum width=3.5cm, minimum height=0.7cm, font=\small\bfseries, below=-\pgflinewidth of fn3] {BCNF};
\end{tikzpicture}
\end{block}
\end{frame}

% SLIDE 3 — Quando 3NF ≠ BCNF: chaves candidatas sobrepostas
\begin{frame}{Por que a 3NF não é suficiente?}
\small
\begin{block}{Situação: múltiplas chaves candidatas sobrepostas}
Considere a relação \textbf{CONSULTA}(\underline{Aluno, Disciplina}, Professor) com:
\begin{itemize}
  \item Cada aluno tem um professor por disciplina: $\{Aluno, Disciplina\} \to Professor$
  \item Cada professor leciona uma única disciplina: $Professor \to Disciplina$
\end{itemize}
Chaves candidatas: $\{Aluno, Disciplina\}$ e $\{Aluno, Professor\}$
\end{block}

\vspace{0.2cm}

\begin{alertblock}{Análise pela 3NF}
A dependência $Professor \to Disciplina$:
\begin{itemize}
  \item $Professor$ \textbf{não} é superchave. \quad \textcolor{red}{\textsf{Violaria 3NF?}}
  \item $Disciplina$ \textbf{é} atributo primo (faz parte da chave $\{Aluno, Disciplina\}$).
  \item Pela 3NF: \textcolor{green!60!black}{\checkmark\ permitido!}
  \item Pela BCNF: \textcolor{red}{\textbf{x proibido!}} ($Professor$ não é superchave)
\end{itemize}
\end{alertblock}

\vspace{0.1cm}
\begin{exampleblock}{Conclusão}
Esta relação está na \textbf{3NF}, mas \underline{não} está na \textbf{BCNF}.
\end{exampleblock}
\end{frame}

  \item O nome da disciplina de cada professor se \textbf{repete} em todos os registros daquele professor.
  \item Mudar o professor de uma disciplina exige atualizar \textbf{todas} as linhas correspondentes.

% SLIDE 5 — Algoritmo de decomposição para BCNF
\begin{frame}{Algoritmo: Decomposição para BCNF}
\small
\begin{block}{Procedimento}
\begin{algorithmic}[1]
\Procedure{DecompõeBCNF}{$R, F$}
  \State $resultado \gets \{R\}$
  \While{existe $S \in resultado$ que \textbf{não} está na BCNF}
    \State Encontra DF $X \to Y$ em $S$ tal que $X$ não é superchave de $S$
    \State $S_1 \gets X \cup Y$ \Comment{Nova tabela com o determinante e dependente}
    \State $S_2 \gets S - Y$ \Comment{Tabela original sem os atributos dependentes}
    \State $resultado \gets (resultado - \{S\}) \cup \{S_1, S_2\}$
  \EndWhile
  \State \Return $resultado$
\EndProcedure
\end{algorithmic}
\end{block}

\vspace{0.2cm}
\begin{alertblock}{Atenção}
\begin{itemize}
  \item A decomposição \textbf{sempre} garante junção sem perdas (\textit{lossless join}).
  \item Mas pode \textbf{perder} algumas dependências funcionais (não preserva $F$).
  \item Em casos assim, pode ser preferível manter a 3NF.
\end{itemize}
\end{alertblock}
\end{frame}

% SLIDE 6 — Decomposição da tabela CONSULTA
\begin{frame}{Decomposição de CONSULTA para BCNF}
\small

\begin{block}{DF que viola BCNF}
$Professor \to Disciplina$ \quad ($Professor$ não é superchave)
\end{block}

\vspace{0.2cm}

\begin{columns}[t]
  \begin{column}{0.48\textwidth}
    \begin{exampleblock}{R1: PROFESSOR\_DISCIPLINA}
    \textbf{Chave:} $\underline{Professor}$
    \vspace{0.1cm}
    \centering
    \begin{tabular}{|l|l|}
    \hline
    \textbf{Professor} & \textbf{Disciplina} \\ \hline
    Prof. Silva & Banco de Dados \\ \hline
    Prof. Lima  & Redes \\ \hline
    Prof. Costa & Algoritmos \\ \hline
    \end{tabular}
    \end{exampleblock}
  \end{column}

  \begin{column}{0.48\textwidth}
    \begin{exampleblock}{R2: ALUNO\_PROFESSOR}
    \textbf{Chave:} $\underline{\{Aluno, Professor\}}$
    \vspace{0.1cm}
    \centering
    \begin{tabular}{|l|l|}
    \hline
    \textbf{Aluno} & \textbf{Professor} \\ \hline
    Ana   & Prof. Silva \\ \hline
    Bruno & Prof. Silva \\ \hline
    Carla & Prof. Silva \\ \hline
    Ana   & Prof. Lima  \\ \hline
    \ldots & \ldots \\ \hline
    \end{tabular}
    \end{exampleblock}
  \end{column}
\end{columns}

\vspace{0.3cm}
\begin{block}{Resultado}
\begin{itemize}
  \item Cada professor aparece \textbf{uma única vez} com sua disciplina.
  \item Ambas as tabelas estão na \textbf{BCNF}.
  \item A DF original $\{Aluno, Disciplina\} \to Professor$ \alert{não} pode ser verificada diretamente (custo da normalização).
\end{itemize}
\end{block}
\end{frame}
