CREATE TABLE tbGerente (
    cod_gerente INTEGER PRIMARY KEY,
    nome_gerente TEXT NOT NULL

CREATE TABLE tbCompanhia (
    cod_companhia INTEGER PRIMARY KEY,
    nome_companhia TEXT NOT NULL,
    cidade_companhia TEXT

CREATE TABLE tbEmpregado (
    cod_emp INTEGER PRIMARY KEY,
    nome_emp TEXT NOT NULL,
    rua_emp TEXT,
    cidade_emp TEXT,
    fone_emp TEXT,
    cod_gerente INTEGER,
    FOREIGN KEY (cod_gerente) REFERENCES tbGerente(cod_gerente)

    \item Escreva o comando \textbf{DDL} para a criação da tabela \textbf{tbTrabalha}, garantindo a integridade referencial com as tabelas criadas anteriormente.\\
\begin{lstlisting}
CREATE TABLE tbTrabalha (
    cod_emp INTEGER,
    cod_companhia INTEGER,
    salario REAL,
    PRIMARY KEY (cod_emp, cod_companhia),
    FOREIGN KEY (cod_emp) REFERENCES tbEmpregado(cod_emp),
    FOREIGN KEY (cod_companhia) REFERENCES tbCompanhia(cod_companhia)
);
\end{lstlisting}

    \item Escreva as consultas SQL solicitadas abaixo:
    \begin{enumerate}[label=\alph*)]
        \item Liste o nome e a cidade de residência de todos os empregados que trabalham para a companhia `Petrobras' e possuem salário superior a R\$ 5.000,00.\\
\begin{lstlisting}
SELECT E.nome_emp, E.cidade_emp
FROM tbEmpregado E
JOIN tbTrabalha T ON E.cod_emp = T.cod_emp
JOIN tbCompanhia C ON T.cod_companhia = C.cod_companhia
WHERE C.nome_companhia = 'Petrobras' AND T.salario > 5000;
\end{lstlisting}

SELECT C.nome_companhia, E.nome_emp
FROM tbCompanhia C
LEFT JOIN tbTrabalha T ON C.cod_companhia = T.cod_companhia
LEFT JOIN tbEmpregado E ON T.cod_emp = E.cod_emp;
\end{lstlisting}
    \end{enumerate}
