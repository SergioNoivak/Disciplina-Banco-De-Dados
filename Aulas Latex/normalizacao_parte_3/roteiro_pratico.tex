\section*{2 \enspace Passo 1 — Limpar o Banco (DROP)}
% ══════════════════════════════════════════════

\begin{atencao}
Execute o bloco \texttt{DROP} \textbf{apenas} quando quiser recomeçar do zero.
A cláusula \texttt{CASCADE} remove automaticamente todas as dependências.
A opção \texttt{IF EXISTS} evita erros se as tabelas ainda não existirem.
\end{atencao}

\vspace{0.3cm}

\begin{lstlisting}[caption={Remoção segura de todas as tabelas}]
DROP TABLE IF EXISTS
    Funcionario_Supervisor,
    Proprietario_Equipamento,
    Ordenha_Equipamento,
    Ordenha_Supervisor,
    Ordenha_Funcionario,
    Vaca_Veterinario,
    Ordenha,
    Vaca,
    Equipamentos,
    Veterinario,
    Funcionario,
    Supervisor,
    Proprietario
CASCADE;
\end{lstlisting}

\begin{dica}
A \textbf{ordem de remoção} importa: tabelas associativas (N:N) devem ser
dropadas \textbf{antes} das tabelas que elas referenciam. Por isso listamos
\texttt{Funcionario\_Supervisor}, \texttt{Vaca\_Veterinario}, etc.\ primeiro.
\end{dica}

\vspace{0.5cm}

% ══════════════════════════════════════════════
\section*{3 \enspace Passo 2 — Criar as Tabelas Principais (DDL)}
% ══════════════════════════════════════════════

\subsection*{3.1 \enspace Entidades independentes}

As tabelas a seguir \textbf{não possuem chaves estrangeiras} e devem ser criadas
primeiro, pois serão referenciadas pelas demais.

\begin{lstlisting}[caption={Proprietário, Equipamentos, Vaca, Veterinário, Funcionário, Supervisor}]
CREATE TABLE Proprietario (
    id               SERIAL PRIMARY KEY,
    nome_completo    VARCHAR(100),
    telefone         VARCHAR(20)
);

CREATE TABLE Equipamentos (
    codigo_equipamento SERIAL PRIMARY KEY,
    tipo               VARCHAR(50),
    ultima_manutencao  DATE,
    situacao           VARCHAR(50)
);

CREATE TABLE Vaca (
    id              SERIAL PRIMARY KEY,
    data_nascimento DATE,
    raca            VARCHAR(50),
    peso            DECIMAL(6,2),
    status_saude    VARCHAR(50)
);

CREATE TABLE Veterinario (
    id                    SERIAL PRIMARY KEY,
    nome_completo         VARCHAR(100),
    telefone              VARCHAR(20),
    registro_profissional VARCHAR(50)
);

CREATE TABLE Funcionario (
    id            SERIAL PRIMARY KEY,
    nome_completo VARCHAR(100),
    telefone      VARCHAR(20),
    turno_periodo VARCHAR(50)
);

CREATE TABLE Supervisor (
    id               SERIAL PRIMARY KEY,
    nome_completo    VARCHAR(100),
    telefone         VARCHAR(20),
    turno_supervisao VARCHAR(50)
);
\end{lstlisting}

\vspace{0.3cm}

\subsection*{3.2 \enspace Tabela fato: Ordenha}

A tabela \texttt{Ordenha} representa o \textbf{evento central} do sistema.
Ela depende de \texttt{Proprietario} (1:N) e \texttt{Vaca} (1:N).
As FKs são adicionadas separadamente com \texttt{ALTER TABLE} para maior clareza.

\begin{lstlisting}[caption={Criação da tabela Ordenha e suas FKs 1:N}]
CREATE TABLE Ordenha (
    id              SERIAL PRIMARY KEY,
    data            DATE,
    horario         TIME,
    quantidade_leite DECIMAL(6,2),
    id_proprietario INT,          -- FK -> Proprietario
    id_vaca         INT NOT NULL  -- FK -> Vaca (obrigatorio)
);

-- Chave estrangeira: Ordenha -> Proprietario (1:N)
ALTER TABLE Ordenha
ADD CONSTRAINT fk_ordenha_proprietario
FOREIGN KEY (id_proprietario) REFERENCES Proprietario(id);

-- Chave estrangeira: Ordenha -> Vaca (1:N)
ALTER TABLE Ordenha
ADD CONSTRAINT fk_ordenha_vaca
FOREIGN KEY (id_vaca) REFERENCES Vaca(id);
\end{lstlisting}

\begin{dica}
Observe que \texttt{id\_vaca} é declarado \texttt{NOT NULL}: toda ordenha
obrigatoriamente pertence a uma vaca. Já \texttt{id\_proprietario} pode ser
nulo --- um evento pode ser registrado antes de vincular um proprietário.
\end{dica}

\vspace{0.5cm}

% ══════════════════════════════════════════════
\section*{4 \enspace Passo 3 — Tabelas Associativas (N:N)}
% ══════════════════════════════════════════════

Relacionamentos N:N são implementados com uma \textbf{tabela intermediária}
cuja chave primária é composta pelas duas FKs envolvidas.

\begin{lstlisting}[caption={Todas as tabelas associativas (N:N)}]
-- Vaca <-> Veterinario
CREATE TABLE Vaca_Veterinario (
    id_veterinario INT NOT NULL,
    id_vaca        INT NOT NULL,
    PRIMARY KEY (id_veterinario, id_vaca),
    FOREIGN KEY (id_veterinario) REFERENCES Veterinario(id),
    FOREIGN KEY (id_vaca)        REFERENCES Vaca(id)
);

-- Ordenha <-> Funcionario
CREATE TABLE Ordenha_Funcionario (
    id_ordenha    INT NOT NULL,
    id_funcionario INT NOT NULL,
    PRIMARY KEY (id_ordenha, id_funcionario),
    FOREIGN KEY (id_ordenha)     REFERENCES Ordenha(id),
    FOREIGN KEY (id_funcionario) REFERENCES Funcionario(id)
);

-- Ordenha <-> Supervisor
CREATE TABLE Ordenha_Supervisor (
    id_ordenha   INT NOT NULL,
    id_supervisor INT NOT NULL,
    PRIMARY KEY (id_ordenha, id_supervisor),
    FOREIGN KEY (id_ordenha)    REFERENCES Ordenha(id),
    FOREIGN KEY (id_supervisor) REFERENCES Supervisor(id)
);

-- Ordenha <-> Equipamentos
CREATE TABLE Ordenha_Equipamento (
    id_ordenha         INT NOT NULL,
    codigo_equipamento INT NOT NULL,
    PRIMARY KEY (id_ordenha, codigo_equipamento),
    FOREIGN KEY (id_ordenha)         REFERENCES Ordenha(id),
    FOREIGN KEY (codigo_equipamento) REFERENCES Equipamentos(codigo_equipamento)
);

-- Proprietario <-> Equipamentos
CREATE TABLE Proprietario_Equipamento (
    id_proprietario    INT NOT NULL,
    codigo_equipamento INT NOT NULL,
    PRIMARY KEY (id_proprietario, codigo_equipamento),
    FOREIGN KEY (id_proprietario)    REFERENCES Proprietario(id),
    FOREIGN KEY (codigo_equipamento) REFERENCES Equipamentos(codigo_equipamento)
);

-- Funcionario <-> Supervisor
CREATE TABLE Funcionario_Supervisor (
    id_funcionario INT NOT NULL,
    id_supervisor  INT NOT NULL,
    PRIMARY KEY (id_funcionario, id_supervisor),
    FOREIGN KEY (id_funcionario) REFERENCES Funcionario(id),
    FOREIGN KEY (id_supervisor)  REFERENCES Supervisor(id)
);
\end{lstlisting}

\vspace{0.3cm}

\section*{6 \enspace Passo 5 — Consultas de Verificação (SELECT)}
% ══════════════════════════════════════════════

Após inserir os dados, execute as consultas abaixo para verificar a integridade.

\begin{lstlisting}[caption={Consultas básicas de verificação}]
-- (a) Todas as ordenhas com nome do proprietario e raca da vaca
SELECT
    o.id          AS ordenha_id,
    o.data,
    o.horario,
    o.quantidade_leite,
    p.nome_completo AS proprietario,
    v.raca
FROM Ordenha o
JOIN Proprietario p ON o.id_proprietario = p.id
JOIN Vaca        v ON o.id_vaca          = v.id
ORDER BY o.data, o.horario;

-- (b) Funcionarios por ordenha
SELECT
    o.id   AS ordenha_id,
    o.data,
    f.nome_completo AS funcionario,
    f.turno_periodo
FROM Ordenha_Funcionario of_
JOIN Ordenha    o ON of_.id_ordenha    = o.id
JOIN Funcionario f ON of_.id_funcionario = f.id
ORDER BY o.id;

-- (c) Quais vacas cada veterinario atende
SELECT
    vet.nome_completo AS veterinario,
    v.raca,
    v.status_saude
FROM Vaca_Veterinario vv
JOIN Veterinario vet ON vv.id_veterinario = vet.id
JOIN Vaca        v   ON vv.id_vaca        = v.id
ORDER BY vet.nome_completo;

-- (d) Total de leite produzido por proprietario
SELECT
    p.nome_completo AS proprietario,
    SUM(o.quantidade_leite) AS total_litros
FROM Ordenha o
JOIN Proprietario p ON o.id_proprietario = p.id
GROUP BY p.nome_completo
ORDER BY total_litros DESC;

-- (e) Equipamentos utilizados em cada ordenha
SELECT
    o.id     AS ordenha_id,
    o.data,
    e.tipo   AS equipamento,
    e.situacao
FROM Ordenha_Equipamento oe
JOIN Ordenha     o ON oe.id_ordenha         = o.id
JOIN Equipamentos e ON oe.codigo_equipamento = e.codigo_equipamento
ORDER BY o.id;
\end{lstlisting}
